import streamlit as st import pandas as pd import requests from streamlit_autorefresh import st_autorefresh # Configuração da tela st.set_page_config( page_title="Painel Alerta Veículos - PMPR", page_icon="🚔", layout="wide" ) # Atualiza a página automaticamente a cada 2 minutos (120.000 ms) st_autorefresh(interval=120000, key="datarefresh") st.title("🚔 Painel Regional de Alerta de Veículos - SESP/PR") st.caption("Atualizado automaticamente a cada 2 minutos | Conectado à Intranet") # Menu Lateral para controle dos municípios st.sidebar.header("⚙️ Configurações da Região") cidades_default = ["LONDRINA", "CAMBÉ", "IBIPORÃ", "ROLÂNDIA", "ARAPONGAS"] cidades_selecionadas = st.sidebar.multiselect( "Selecione os municípios monitorados:", options=[ "LONDRINA", "CAMBÉ", "IBIPORÃ", "ROLÂNDIA", "ARAPONGAS", "MARINGÁ", "CURITIBA", "PONTA GROSSA", "CASCAVEL", "FOZ DO IGUAÇU" ], default=cidades_default ) # Função para ler a tabela do Alerta Veículos SESP @st.cache_data(ttl=110) def buscar_dados_sesp(): url = "https://alertaveiculos.sesp.parana/alertaveiculos/index.do?action=iniciarProcesso" headers = {"User-Agent": "Mozilla/5.0"} try: res = requests.get(url, headers=headers, timeout=10) tabelas = pd.read_html(res.text) return tabelas[0] if tabelas else pd.DataFrame() except Exception as e: st.error(f"Erro ao conectar com a SESP Intranet: {e}") return pd.DataFrame() df = buscar_dados_sesp() if not df.empty: # Procura a coluna de município/cidade na tabela da SESP col_cidade = [c for c in df.columns if "MUNICIPIO" in str(c).upper() or "CIDADE" in str(c).upper()] if col_cidade: col_nome = col_cidade[0] cidades_upper = [c.upper() for c in cidades_selecionadas] # Aplica o filtro regional df_filtrado = df[df[col_nome].astype(str).str.upper().isin(cidades_upper)] # Cartões no topo c1, c2, c3 = st.columns(3) c1.metric("Alertas no Estado", len(df)) c2.metric("Ocorrências na Sua Região", len(df_filtrado)) c3.metric("Cidades Monitoradas", len(cidades_selecionadas)) st.divider() if not df_filtrado.empty: st.error("🚨 ATENÇÃO: VEÍCULOS ROUBADOS/FURTADOS NA SUA REGIÃO!") st.dataframe(df_filtrado, use_container_width=True) else: st.success("✅ Nenhuma ocorrência registrada para as cidades selecionadas no momento.") else: st.write("Tabela carregada:", df) Passo 5: Iniciar o Servidor e Descobrir o IP No Prompt de Comando, navegue até a pasta: cd C:\AlertaVeiculos Inicie o Streamlit: streamlit run app.py O terminal mostrará duas URLs: Local URL: http://localhost:8501 Network URL: http://10.x.x.x:8501 (Copie este endereço de rede!)
